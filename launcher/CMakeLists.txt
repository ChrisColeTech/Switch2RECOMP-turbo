cmake_minimum_required(VERSION 3.21)
project(SwitchRecompLauncher VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets Test)

add_library(launcher_core STATIC library.cpp library.h buildmanager.cpp buildmanager.h processwatch.cpp processwatch.h)
target_link_libraries(launcher_core PUBLIC Qt6::Core)
if(WIN32)
    target_link_libraries(launcher_core PRIVATE user32)
endif()
target_include_directories(launcher_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
qt_add_executable(SwitchRecompLauncher WIN32 main.cpp window.cpp window.h artwork.cpp artwork.h theme.cpp theme.h filterpopup.cpp filterpopup.h resources.qrc)
target_link_libraries(SwitchRecompLauncher PRIVATE launcher_core Qt6::Widgets)
if(MSVC)
    target_compile_options(launcher_core PRIVATE /W4 /utf-8)
    target_compile_options(SwitchRecompLauncher PRIVATE /W4 /utf-8)
endif()
enable_testing()
qt_add_executable(launcher-tests tests.cpp theme.cpp theme.h)
target_link_libraries(launcher-tests PRIVATE launcher_core Qt6::Test Qt6::Widgets)
add_test(NAME launcher-core COMMAND launcher-tests)
qt_add_executable(launcher-ui-tests gui_tests.cpp filterpopup.cpp filterpopup.h theme.cpp theme.h artwork.cpp artwork.h resources.qrc)
target_link_libraries(launcher-ui-tests PRIVATE launcher_core Qt6::Test Qt6::Widgets)
add_test(NAME launcher-filters COMMAND launcher-ui-tests -platform offscreen)
install(TARGETS SwitchRecompLauncher RUNTIME DESTINATION .)
